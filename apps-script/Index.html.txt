  function callServer(fn, authToken, args, timeoutMs = 15000) {
    return new Promise((resolve, reject) => {
      let finished = false;

      const timer = setTimeout(() => {
        if (finished) return;
        finished = true;
        reject(new Error('SERVER_TIMEOUT'));
      }, timeoutMs);

      google.script.run
        .withSuccessHandler(result => {
          if (finished) return;
          finished = true;
          clearTimeout(timer);
          resolve(result);
        })
        .withFailureHandler(err => {
          if (finished) return;
          finished = true;
          clearTimeout(timer);
          reject(new Error(err?.message || String(err)));
        })
        .api(fn, authToken || '', args || []);
    });
  }

  function publicGas(fn, ...args) {
    return callServer(fn, '', args);
  }

  function gas(fn, ...args) {
    return callServer(fn, state.authToken, args)
      .catch(err => {
        const message = err?.message || String(err);
        if (message.includes('AUTH_REQUIRED')) {
          handleAuthExpired();
        }
        throw err;
      });

      // a first-time PIN setup, or a slow backend can never leave a blank spinner.
      showLogin();

      let savedToken = '';
      try {
        savedToken = localStorage.getItem(AUTH_STORAGE_KEY) || '';
      } catch (_) {}

      if (savedToken) {
        try {
          const check = await publicGas('checkSession', savedToken);
          if (check && check.valid) {
            state.authToken = savedToken;
            await loadJournal();
            notifyParentReady();
            return;
          }
        } catch (err) {
          console.warn('Session check failed', err);
          const message = String((err && err.message) || err);

          if (message.includes('SERVER_TIMEOUT')) {
            $('#loginMessage').textContent =
              'Google Apps Script did not answer the session check. You can still try your PIN below.';
          }
        }
      }

      try { localStorage.removeItem(AUTH_STORAGE_KEY); } catch (_) {}

      // This is informational only; the login screen is already usable.

        const authStatus = await publicGas('getAuthStatus');
        if (authStatus && authStatus.pinConfigured === false) {
          const subtitle = $('#loginSubtitle');

            'PIN setup is required before the journal can be unlocked.';

      } catch (err) {

      notifyParentReady();

    } catch (err) {
      console.error('Journal initialization failed', err);
      showStartupFailure(
        'Journal initialization error: ' + String((err && err.message) || err)

      notifyParentReady();
    }

  function setupPhotoObserver() {
    if (!('IntersectionObserver' in window)) return;
    state.observer = new IntersectionObserver(entries => {
      entries.forEach(x => {
        if (x.isIntersecting) {
          state.observer.unobserve(x.target);
          loadPhotoInto(x.target, x.target.dataset.photoId);
        }
      });
    }, { rootMargin:'250px' });
  }

  function observePhotos() {
    $$('.lazyPhoto').forEach(el => {
      if (el.dataset.observed) return;
      el.dataset.observed = '1';
      if (state.observer) state.observer.observe(el);
      else loadPhotoInto(el, el.dataset.photoId);
    });
  }

  async function loadPhotoInto(el, id) {
    if (!id || !el?.isConnected) return;
    try {
      const data = await getPhotoData(id);
      const img = document.createElement('img');
      img.className = el.classList.contains('cardPhoto') ? 'cardPhoto' : '';
      img.alt = '';
      img.loading = 'lazy';
      img.src = data.dataUrl;
      el.replaceWith(img);
    } catch (_) {
      el.textContent = 'Photo unavailable';
    }
  }

  async function getPhotoData(id) {
    if (state.photoCache.has(id)) return state.photoCache.get(id);
    const data = await gas('getAttachmentData', id, true);
    state.photoCache.set(id, data);
    return data;
  }
